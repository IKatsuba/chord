// Scratch probe for CHD-172's runtime spike; never in the tree.
import type { CliRenderer } from '@opentui/core';
export const probe = (globalThis as any).__probe ??= { log: [] as string[], renderer: undefined as CliRenderer | undefined, copies: [] as string[], notes: [] as string[], impl: undefined as undefined | ((t: string, s: AbortSignal) => Promise<string>) };
export function state(r: CliRenderer | undefined): string {
  const s = r?.getSelection();
  if (!s) return `none(has=${r?.hasSelection})`;
  return `${s.isStart ? 'start,' : ''}${s.isDragging ? 'dragging' : 'done'}:${JSON.stringify(s.getSelectedText())}`;
}
export function note(what: string) { probe.log.push(`${what} | ${state(probe.renderer)}`); }
let generation = 0;
let inFlight: AbortController | undefined;
export function copy(text: string) {
  inFlight?.abort();
  const mine = ++generation;
  const ac = new AbortController();
  inFlight = ac;
  probe.copies.push(text);
  const raw = probe.impl ? probe.impl(text, ac.signal) : Promise.resolve('written');
  let settled = false;
  const p = Promise.race([raw, new Promise<string>((res) => setTimeout(() => res('chord-bound-timeout'), 1200))]);
  raw.then((r) => { if (settled) probe.log.push(`late result dropped after the bound: ${r}`); });
  p.then((r) => { settled = true; return r; }).then((r) => { if (mine === generation && !ac.signal.aborted) probe.notes.push(`copied ${text.length} (${r})`); else probe.log.push(`late result dropped: ${r} for gen ${mine}`); },
         (e) => { if (mine === generation) probe.notes.push(`failed ${e}`); });
}
export function disposeCopies() { inFlight?.abort(); generation++; }
let last = '';
export function watchFrames(r: CliRenderer) {
  if ((r as any).__watched) return;
  (r as any).__watched = true;
  r.on('destroy', () => { probe.log.push('renderer destroy: copies disposed'); disposeCopies(); });
  r.on('frame', () => {
    const s = r.getSelection();
    const now = s === null ? 'none' : s.getSelectedText() === '' ? 'empty' : 'text';
    if (now !== 'text') hold.on = false;
    if (now !== last) { probe.log.push(`frame event: selection ${last || '?'} -> ${now}`); last = now; }
  });
}
export const hold = { on: false, at: undefined as number | undefined, enabled: true };
export function pinned(first: number): number {
  if (!hold.on || !hold.enabled) { hold.at = undefined; return first; }
  hold.at ??= first;
  return Math.min(hold.at, first);
}

// No real clipboard: the host is a stub, the terminal adapter records instead of writing.
import { createClipboard } from '@opentui/core';

const osc: string[] = [];
const terminal = {
  remote: false,
  writeText: (t: string) => {
    osc.push(t);
    return { status: 'attempted', capability: 'unknown' } as const;
  },
  clear: () => ({ status: 'attempted', capability: 'unknown' }) as const,
};
type Mode =
  | 'written'
  | 'failed'
  | 'unsupported'
  | 'timed-out'
  | 'hang-honours-abort'
  | 'hang-ignores-abort'
  | 'slow-200ms';
const stub = (mode: Mode): any => ({
  maxWriteBytes: 8 << 20,
  read: async () => ({ status: 'unsupported' }),
  writeText: (_t: string, o: { signal?: AbortSignal }) => {
    if (mode === 'hang-ignores-abort') return new Promise(() => {});
    if (mode === 'hang-honours-abort')
      return new Promise((res) =>
        o.signal?.addEventListener('abort', () => res({ status: 'cancelled' })),
      );
    if (mode === 'slow-200ms')
      return new Promise((res) => setTimeout(() => res({ status: 'written' }), 200));
    return Promise.resolve(
      mode === 'failed' ? { status: 'failed', error: new Error('no provider') } : { status: mode },
    );
  },
  clear: async () => ({ status: 'unsupported' }),
  dispose: async () => {},
});
const race = <T>(p: Promise<T>, ms: number) =>
  Promise.race([
    p.then((v) => ({ v })),
    new Promise<{ timeout: true }>((r) => setTimeout(() => r({ timeout: true }), ms)),
  ]);

for (const mode of [
  'written',
  'failed',
  'unsupported',
  'timed-out',
  'hang-honours-abort',
  'hang-ignores-abort',
] as Mode[]) {
  osc.length = 0;
  const svc = createClipboard({ host: stub(mode), terminal });
  const ac = new AbortController();
  setTimeout(() => ac.abort(), 300);
  const t0 = performance.now();
  const r = await race(svc.writeText('hello', { destination: 'best-available', signal: ac.signal }), 1000);
  console.log(
    `local, host ${mode}: ${'timeout' in r ? 'still pending at 1000 ms (abort at 300 ms did not settle it)' : `host=${r.v.host.status} terminal=${r.v.terminal.status}`} after ${(performance.now() - t0).toFixed(0)} ms; OSC 52 fallback written: ${osc.length}`,
  );
  const d0 = performance.now();
  const d = await race(svc.dispose(), 1000);
  console.log(`   dispose: ${'timeout' in d ? 'still pending at 1000 ms' : 'settled'} after ${(performance.now() - d0).toFixed(0)} ms`);
}

{
  osc.length = 0;
  const svc = createClipboard({ host: stub('slow-200ms'), terminal });
  const a = new AbortController();
  const first = svc.writeText('first', { destination: 'best-available', signal: a.signal });
  a.abort();
  const second = svc.writeText('second', { destination: 'best-available' });
  const [r1, r2] = await Promise.all([first, second]);
  console.log(
    `superseded (first aborted at once, host ignores the signal for 200 ms): first host=${r1.host.status} terminal=${r1.terminal.status}; second host=${r2.host.status}; OSC 52 written ${osc.length}`,
  );
}
{
  const svc = createClipboard({ host: stub('hang-honours-abort'), terminal });
  const p = svc.writeText('x', { destination: 'best-available' });
  const d0 = performance.now();
  await svc.dispose();
  const r = await p;
  console.log(`pending at dispose (host honours abort): dispose ${(performance.now() - d0).toFixed(0)} ms; the write settled host=${r.host.status}`);
  try {
    await svc.writeText('y', { destination: 'best-available' });
  } catch (e) {
    console.log(`   a write after dispose rejects: ${(e as Error).message}`);
  }
}
