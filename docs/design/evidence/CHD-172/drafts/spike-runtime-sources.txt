async function capture(label: string, opts: Record<string, unknown>) {
  const log: string[] = [];
  const stdin = Object.assign(new Readable({ read() {} }), { isTTY: true, setRawMode() { return stdin; } });
  const stdout = Object.assign(new Writable({ write(c, _e, d) { log.push(String(c)); d(); } }),
    { isTTY: true, columns: 80, rows: 24, getColorDepth: () => 24 });
  const r = await createCliRenderer({ stdin: stdin as any, stdout: stdout as any, exitOnCtrlC: false, exitSignals: [], consoleMode: 'disabled', openConsoleOnError: false, ...opts });
  await new Promise((res) => setTimeout(res, 150));
  const start = log.join('');
  log.length = 0;
  r.destroy();
  await r.closed.catch(() => {});
  const end = log.join('');
  const modes = (s: string) => [...s.matchAll(/\x1b\[\?([0-9;]+)([hl])/g)].map((m) => `?${m[1]}${m[2]}`).join(' ');
  console.log(`${label}\n  start: ${modes(start)}\n  end:   ${modes(end)}`);
}

// Every write goes to a fake stdout: the real clipboard is never touched.
import { Readable, Writable } from 'node:stream';
import { createCliRenderer, createClipboard, createRendererClipboardAdapter, TextRenderable } from '@opentui/core';

async function open(env: Record<string, string | undefined>) {
  const saved = { ...process.env };
  Object.assign(process.env, env);
  for (const [k, v] of Object.entries(env)) if (v === undefined) delete process.env[k];
  const log: string[] = [];
  const stdin = Object.assign(new Readable({ read() {} }), { isTTY: true, setRawMode() { return stdin; } });
  const stdout = Object.assign(new Writable({ write(c, _e, d) { log.push(String(c)); d(); } }),
    { isTTY: true, columns: 80, rows: 24, getColorDepth: () => 24 });
  const r = await createCliRenderer({ stdin: stdin as any, stdout: stdout as any, exitOnCtrlC: false, exitSignals: [], consoleMode: 'disabled', openConsoleOnError: false, useMouse: true });
  process.env = saved;
  await new Promise((res) => setTimeout(res, 150));
  log.length = 0;
  return { r, log, stdin };
}
const osc = (s: string) => [...s.matchAll(/(\x1bP[^\x1b]*)?\x1b\]52;([^;]*);([^\x07\x1b]*)(\x07|\x1b\\)/g)];
