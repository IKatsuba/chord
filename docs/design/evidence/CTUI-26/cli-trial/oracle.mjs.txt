for(const argv of vectors){let value;try{await chordCommand().noExit().throwErrors().action((options,agent)=>{value={...(agent===undefined?{}:{agent}),...(options.resume===undefined?{}:{resume:options.resume}),...(options.continue===undefined?{}:{continue:options.continue}),...(options.agentCommand===undefined?{}:{agentCommand:options.agentCommand})};}).parse(argv);cases.push({argv,value:value??'no action'});}catch(error){cases.push({argv,error:error.message});}}
